--- source (compiled inside the container):
#include <gtksourceview/gtksource.h>
#include <stdio.h>
int main(void) {
  GtkSourceBuffer *b = gtk_source_buffer_new(NULL);
  GtkTextIter it;
  gtk_text_buffer_get_start_iter(GTK_TEXT_BUFFER(b), &it);
  gtk_text_buffer_insert(GTK_TEXT_BUFFER(b), &it, "hello", -1);
  GtkSourceUndoManager *m = gtk_source_buffer_get_undo_manager(b);
  printf("manager=%p can_undo=%d\n", (void*)m, gtk_source_buffer_can_undo(b));
  gtk_source_buffer_undo(b);
  GtkTextIter s, e; gtk_text_buffer_get_bounds(GTK_TEXT_BUFFER(b), &s, &e);
  char *t = gtk_text_buffer_get_text(GTK_TEXT_BUFFER(b), &s, &e, TRUE);
  printf("after undo text=[%s] can_undo=%d can_redo=%d\n", t, gtk_source_buffer_can_undo(b), gtk_source_buffer_can_redo(b));
  gtk_source_buffer_redo(b);
  gtk_text_buffer_get_bounds(GTK_TEXT_BUFFER(b), &s, &e);
  t = gtk_text_buffer_get_text(GTK_TEXT_BUFFER(b), &s, &e, TRUE);
  printf("after redo text=[%s]\n", t);
  return 0;
}

## Probe 3: the buffer records its initial load as an undo action
$ docker run --rm vedit-build:local ... gtk_text_buffer_set_text(...) then can_undo
observed: after-load can_undo=1; after an edit inside a user action: undo1 -> loaded text, undo2 -> empty buffer.
=> Document exposes only its own edits through can_undo/undo; the load action stays in the
   native stack but is never reachable (a freshly opened document cannot be undone to empty).
