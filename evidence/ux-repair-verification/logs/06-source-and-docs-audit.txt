bool ProjectModel::importFilelists(const QStringList &paths, QString *error, int *addedFilelists)
{
    if (addedFilelists != nullptr) {
        *addedFilelists = 0;
    }
    if (state_.root.isEmpty()) {
        if (error != nullptr) {
            *error = QStringLiteral("open a folder before importing a filelist");
        }
        return false;
    }
    QStringList accepted = state_.filelists;
    for (const QString &path : paths) {
        const QFileInfo info(path);
        if (!info.exists() || !info.isFile()) {
            if (error != nullptr) {
                *error = QStringLiteral("%1 is not a readable filelist").arg(path);
            }
            return false; // atomic: nothing is added on a bad entry
        }
        const QString absolute = info.absoluteFilePath();
        if (!accepted.contains(absolute)) {
            accepted.append(absolute);
        }
    }
    if (error != nullptr) {
        error->clear();
    }
    if (accepted == state_.filelists) {
        return true; // already imported: no re-parse needed
    }
    if (addedFilelists != nullptr) {
        *addedFilelists =
            static_cast<int>(accepted.size() - state_.filelists.size());
    }
    state_.filelists = accepted;
    saveState();
    emit filelistsChanged();
    startExpansion();
    return true;
}

    if (model_ == nullptr) {

ProjectService::Start ProjectService::start(const QString &projectRoot, const QStringList &filelists,
                                            const QStringList &variables, const QString &settingsPath,
                                            quint64 *job)
{
    if (service_ == nullptr || job == nullptr) {
        return Start::Rejected;
    }
    const QByteArray path = projectRoot.toUtf8();
    const QByteArray settings = settingsPath.toUtf8();
    BorrowedList lists(filelists);
    BorrowedList table(variables);
    const quint64 id = static_cast<quint64>(vd_project_expand_v2(
        service_, path.constData(), &lists.list,
        settings.isEmpty() ? nullptr : settings.constData(), &table.list));
    if (id == 0) {
        return Start::Rejected;
    }
    *job = id;
    return Start::Started;
}

### 路径显示（统一规则）

Qt6 产品里，**下列界面**显示路径时用同一条规则（`src/common/path_display.h`）：

- 工程根存在且文件在根内 → 显示相对工程根的路径（不带 `./`）；
- 否则 → 显示绝对路径。

覆盖编辑器标签的 tooltip、窗口标题、Files 树的 tooltip、Project 面板的文件与 filelist 列表、
`Ctrl+P` 文件面板和搜索结果行。**打开和去重仍用绝对路径**，显示规则不改变任何打开行为；
工程根会先规范化（绝对路径、去掉 `.` / `..` / 尾斜杠）再参与比较。编辑器标签文字本身显示文件名
（不是路径），tooltip 才显示上面这条规则算出的路径。

**状态栏 / 面板的一次性确认与错误提示是例外**：palette / Files 树 / 搜索跳转的 `Opened …`、
Project 面板的 `Removed …` 和错误提示仍显示绝对路径（只停留几秒，不是路径视图）。除上面枚举的
界面和这些一次性提示外，如果看到绝对路径显示，就是 bug。

### 文件列表（`.f`）里的变量与转义

在 Qt6 产品里，打开一个 `.f` / `.F` / `.flist` 会**同时**把它作为工程 filelist 导入（规则见上文
"工程根的起点"）；filelist 也仍然作为普通文本标签打开。**编辑后要按 Project 面板的 `Refresh`
重新展开**（重新打开同一个 filelist 不会重新解析），重复打开同一个 filelist 是幂等的：不会重复
加入，也不会重复解析，Project 面板底部会提示 `Already imported: nothing changed`。解析诊断（缺失的 `+incdir+`、未知 token、
嵌套 `-f` 问题等）显示在 Project 面板的诊断列表里，不会弹出模态框。
