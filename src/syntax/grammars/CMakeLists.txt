# VEdit grammar lane (syn-grammars, task T17).
#
# Binds the vendored tree-sitter-verilog grammar (MIT; provenance in
# third_party/tree-sitter/PROVENANCE.md) into the T15 SyntaxProvider seam and
# embeds the three logical query sets (highlights / folds / structure) from
# resources/queries/.
#
# Compile-time switching (C15): changing the grammar id or turning a grammar off
# is a CONFIGURATION change. No tracked source file changes — the identity, the
# licence, the queries and the manifest are generated into the build directory.
#
#   VEDIT_SYNTAX_GRAMMAR        none | verilog | systemverilog   (baked-in id)
#   VEDIT_GRAMMAR_VERILOG       ON/OFF  compile + register the Verilog binding
#   VEDIT_GRAMMAR_SYSTEMVERILOG ON/OFF  compile + register the SystemVerilog binding
#   VEDIT_GRAMMAR_BUILD_TESTS   ON/OFF  build the lane validator (ctest)
#
# Standalone:
#   cmake -S src/syntax/grammars -B target/syn-grammars -G Ninja \
#         -DVEDIT_SYNTAX_GRAMMAR=verilog
#   cmake --build target/syn-grammars --parallel && ctest --test-dir target/syn-grammars
cmake_minimum_required(VERSION 3.20)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vedit_syn_grammars LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_C_STANDARD 11)
  set(CMAKE_C_STANDARD_REQUIRED ON)
  enable_testing()
endif()

option(VEDIT_GRAMMAR_VERILOG "Compile + register the Verilog grammar binding" ON)
option(VEDIT_GRAMMAR_SYSTEMVERILOG "Compile + register the SystemVerilog grammar binding" ON)
option(VEDIT_GRAMMAR_BUILD_TESTS "Build the grammar lane validator (needs the tree-sitter backend)" ON)
set(VEDIT_SYNTAX_GRAMMAR "verilog" CACHE STRING
    "Grammar binding id baked into this build (none|verilog|systemverilog)")

# The provider lane owns the seam; when this directory is the top-level build,
# pull it in (same pattern as the root build for src/app).
if(NOT TARGET vedit_syntax_provider)
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../provider" "${CMAKE_CURRENT_BINARY_DIR}/provider")
endif()

if(NOT VEDIT_SYNTAX_GRAMMAR MATCHES "^(none|verilog|systemverilog)$")
  message(FATAL_ERROR
    "VEDIT_SYNTAX_GRAMMAR must be 'none', 'verilog' or 'systemverilog', got '${VEDIT_SYNTAX_GRAMMAR}'")
endif()
if(VEDIT_SYNTAX_GRAMMAR STREQUAL "verilog" AND NOT VEDIT_GRAMMAR_VERILOG)
  message(FATAL_ERROR "VEDIT_SYNTAX_GRAMMAR=verilog but VEDIT_GRAMMAR_VERILOG=OFF")
endif()
if(VEDIT_SYNTAX_GRAMMAR STREQUAL "systemverilog" AND NOT VEDIT_GRAMMAR_SYSTEMVERILOG)
  message(FATAL_ERROR "VEDIT_SYNTAX_GRAMMAR=systemverilog but VEDIT_GRAMMAR_SYSTEMVERILOG=OFF")
endif()

set(VEDIT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(VEDIT_TS_VERILOG_DIR "${VEDIT_REPO_ROOT}/third_party/tree-sitter/verilog")
set(VEDIT_QUERY_VERILOG_DIR "${VEDIT_REPO_ROOT}/resources/queries/verilog")
set(VEDIT_QUERY_SYSTEMVERILOG_DIR "${VEDIT_REPO_ROOT}/resources/queries/systemverilog")
set(VEDIT_GRAMMAR_VERSION "4457145e795b363f072463e697dfe2f6973c9a52")
set(VEDIT_GRAMMAR_LICENSE "MIT")

foreach(_f
    "${VEDIT_TS_VERILOG_DIR}/src/parser.c"
    "${VEDIT_TS_VERILOG_DIR}/src/tree_sitter/parser.h"
    "${VEDIT_TS_VERILOG_DIR}/LICENSE")
  if(NOT EXISTS "${_f}")
    message(FATAL_ERROR "missing vendored grammar file: ${_f} (see third_party/tree-sitter/PROVENANCE.md)")
  endif()
endforeach()

if(VEDIT_GRAMMAR_VERILOG)
  set(_verilog_on "true")
  set(_verilog_define "1")
else()
  set(_verilog_on "false")
  set(_verilog_define "0")
endif()
if(VEDIT_GRAMMAR_SYSTEMVERILOG)
  set(_sv_on "true")
  set(_sv_define "1")
else()
  set(_sv_on "false")
  set(_sv_define "0")
endif()

# ---- query sources: read at configure time, embedded as build artifacts ----
foreach(_lang verilog systemverilog)
  string(TOUPPER "${_lang}" _lang_upper)
  foreach(_q highlights folds structure)
    if(NOT EXISTS "${VEDIT_QUERY_${_lang_upper}_DIR}/${_q}.scm")
      message(FATAL_ERROR
        "missing query file: ${VEDIT_QUERY_${_lang_upper}_DIR}/${_q}.scm (resources/queries/${_lang})")
    endif()
  endforeach()
endforeach()
file(READ "${VEDIT_QUERY_VERILOG_DIR}/highlights.scm" _v_highlights)
file(READ "${VEDIT_QUERY_VERILOG_DIR}/folds.scm" _v_folds)
file(READ "${VEDIT_QUERY_VERILOG_DIR}/structure.scm" _v_structure)
file(READ "${VEDIT_QUERY_SYSTEMVERILOG_DIR}/highlights.scm" _sv_highlights)
file(READ "${VEDIT_QUERY_SYSTEMVERILOG_DIR}/folds.scm" _sv_folds)
file(READ "${VEDIT_QUERY_SYSTEMVERILOG_DIR}/structure.scm" _sv_structure)

# ---- generated configuration header (build artifact, never a source edit) ----
set(VEDIT_GRAMMAR_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${VEDIT_GRAMMAR_GEN_DIR}")

set(_header
"// Generated by CMake from VEDIT_GRAMMAR_* / VEDIT_SYNTAX_GRAMMAR and
// resources/queries/*.scm. This file is a BUILD ARTIFACT: switching grammar
// rewrites it; no tracked source file changes (C15).
#pragma once

#define VEDIT_GRAMMAR_VERILOG_ENABLED ${_verilog_define}
#define VEDIT_GRAMMAR_SYSTEMVERILOG_ENABLED ${_sv_define}

namespace vedit::syntax::generated {

struct GrammarInfo {
    const char *id;
    const char *display;
    const char *version;
    const char *license;
    bool enabled;
};

inline constexpr GrammarInfo kVerilog = {\"verilog\", \"Verilog (tree-sitter-verilog)\",
    \"${VEDIT_GRAMMAR_VERSION}\", \"${VEDIT_GRAMMAR_LICENSE}\", ${_verilog_on}};
inline constexpr GrammarInfo kSystemVerilog = {\"systemverilog\",
    \"SystemVerilog (tree-sitter-verilog)\", \"${VEDIT_GRAMMAR_VERSION}\",
    \"${VEDIT_GRAMMAR_LICENSE}\", ${_sv_on}};

inline constexpr const char *kVerilogHighlights = R\"VEDITQ(${_v_highlights})VEDITQ\";
inline constexpr const char *kVerilogFolds = R\"VEDITQ(${_v_folds})VEDITQ\";
inline constexpr const char *kVerilogStructure = R\"VEDITQ(${_v_structure})VEDITQ\";
inline constexpr const char *kSystemVerilogHighlights = R\"VEDITQ(${_sv_highlights})VEDITQ\";
inline constexpr const char *kSystemVerilogFolds = R\"VEDITQ(${_sv_folds})VEDITQ\";
inline constexpr const char *kSystemVerilogStructure = R\"VEDITQ(${_sv_structure})VEDITQ\";

}  // namespace vedit::syntax::generated
")
file(WRITE "${VEDIT_GRAMMAR_GEN_DIR}/grammars_config_generated.h" "${_header}")

# ---- build manifest: which grammars are compiled in, with licence+queries ----
file(SHA256 "${VEDIT_QUERY_VERILOG_DIR}/highlights.scm" _v_high_sha)
file(SHA256 "${VEDIT_QUERY_VERILOG_DIR}/folds.scm" _v_folds_sha)
file(SHA256 "${VEDIT_QUERY_VERILOG_DIR}/structure.scm" _v_struct_sha)
file(SHA256 "${VEDIT_QUERY_SYSTEMVERILOG_DIR}/highlights.scm" _sv_high_sha)
file(SHA256 "${VEDIT_QUERY_SYSTEMVERILOG_DIR}/folds.scm" _sv_folds_sha)
file(SHA256 "${VEDIT_QUERY_SYSTEMVERILOG_DIR}/structure.scm" _sv_struct_sha)
file(SHA256 "${VEDIT_TS_VERILOG_DIR}/src/parser.c" _parser_sha)

set(_manifest
"# VEdit grammar build manifest (generated at configure time, task T17)
VEDIT_SYNTAX_GRAMMAR=${VEDIT_SYNTAX_GRAMMAR}
grammar=verilog enabled=${_verilog_define} license=${VEDIT_GRAMMAR_LICENSE} version=${VEDIT_GRAMMAR_VERSION} source=third_party/tree-sitter/verilog parser_sha256=${_parser_sha}
query=resources/queries/verilog/highlights.scm sha256=${_v_high_sha}
query=resources/queries/verilog/folds.scm sha256=${_v_folds_sha}
query=resources/queries/verilog/structure.scm sha256=${_v_struct_sha}
grammar=systemverilog enabled=${_sv_define} license=${VEDIT_GRAMMAR_LICENSE} version=${VEDIT_GRAMMAR_VERSION} source=third_party/tree-sitter/verilog parser_sha256=${_parser_sha}
query=resources/queries/systemverilog/highlights.scm sha256=${_sv_high_sha}
query=resources/queries/systemverilog/folds.scm sha256=${_sv_folds_sha}
query=resources/queries/systemverilog/structure.scm sha256=${_sv_struct_sha}
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/grammars_manifest.txt" "${_manifest}")
message(STATUS "syn-grammars: compiled grammar id='${VEDIT_SYNTAX_GRAMMAR}' verilog=${_verilog_define} systemverilog=${_sv_define}")

# ---- grammar + binding library ----
if(VEDIT_GRAMMAR_VERILOG OR VEDIT_GRAMMAR_SYSTEMVERILOG)
  add_library(vedit_tree_sitter_verilog STATIC "${VEDIT_TS_VERILOG_DIR}/src/parser.c")
  target_include_directories(vedit_tree_sitter_verilog PUBLIC "${VEDIT_TS_VERILOG_DIR}/src")
endif()

add_library(vedit_syntax_grammars STATIC bindings.cpp)
target_include_directories(vedit_syntax_grammars PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${VEDIT_GRAMMAR_GEN_DIR}")
target_link_libraries(vedit_syntax_grammars PUBLIC vedit_syntax_provider)
if(VEDIT_GRAMMAR_VERILOG OR VEDIT_GRAMMAR_SYSTEMVERILOG)
  target_link_libraries(vedit_syntax_grammars PUBLIC vedit_tree_sitter_verilog)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(vedit_syntax_grammars PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

# ---- lane validator ----
if(VEDIT_GRAMMAR_BUILD_TESTS AND VEDIT_SYNTAX_TREESITTER)
  add_executable(vedit_syn_grammars_tests tests/test_syn_grammars.cpp)
  target_link_libraries(vedit_syn_grammars_tests PRIVATE vedit_syntax_grammars)
  target_compile_definitions(vedit_syn_grammars_tests PRIVATE
    VEDIT_EXPECT_COMPILED_GRAMMAR="${VEDIT_SYNTAX_GRAMMAR}"
    VEDIT_EXPECT_VERILOG_ENABLED=${_verilog_define}
    VEDIT_EXPECT_SYSTEMVERILOG_ENABLED=${_sv_define}
    VEDIT_GRAMMAR_SAMPLES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(vedit_syn_grammars_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME vedit-syn-grammars-tests COMMAND vedit_syn_grammars_tests)
  set_tests_properties(vedit-syn-grammars-tests PROPERTIES TIMEOUT 120)
elseif(VEDIT_GRAMMAR_BUILD_TESTS)
  message(STATUS "syn-grammars: tests skipped (tree-sitter backend not compiled)")
endif()
