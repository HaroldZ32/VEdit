# VEdit SyntaxProvider (syn-provider lane, task T15).
#
# Usable two ways:
#   * standalone:  cmake -S src/syntax/provider -B <build> -G Ninja && ctest
#   * from the root build (once the build base wires it in, like src/app):
#       add_subdirectory(src/syntax/provider)
#
# Compile-time switches (contract C15/C18):
#   VEDIT_SYNTAX_GRAMMAR     none | external   grammar binding id baked into the build
#   VEDIT_SYNTAX_TREESITTER  ON/OFF            compile the tree-sitter backend at all
#   VEDIT_SYNTAX_TS_LINK     shared|static     how libtree-sitter is linked (C18)
# Switching grammar = changing configuration + registrations, never this tree's
# interface/callers. The runtime master switch lives in SyntaxService.
cmake_minimum_required(VERSION 3.20)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vedit_syntax_provider LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  enable_testing()
endif()

option(VEDIT_SYNTAX_TREESITTER "Compile the tree-sitter backend" ON)

# ---- grammar id ownership (AC-S1.2 / G3) -----------------------------------
# The id names a grammar that a GRAMMAR lane registers, and the root build adds
# src/syntax/grammars AFTER this directory. Seeding the cache here therefore made
# this file's "none" win over that lane's "verilog" and silently baked a product
# with no grammar (observed before T12: CMakeCache.txt had
# VEDIT_SYNTAX_GRAMMAR=none and grammar_config_generated.h kCompiledGrammar
# "none"). Rules now:
#   * standalone provider build: no grammar lane exists -> default "none";
#   * product build: no default is seeded here. The grammar lane (or an explicit
#     -DVEDIT_SYNTAX_GRAMMAR=...) owns the value, and this file reads it at the
#     END of the top-level configure (cmake_language(DEFER ...) below), after
#     every lane has had its say. An EMPTY value is a hard error: silently
#     falling back to "none" is exactly the defect this fixes.
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(VEDIT_SYNTAX_GRAMMAR_OWNER "src/syntax/grammars"
      CACHE INTERNAL "lane that owns VEDIT_SYNTAX_GRAMMAR (product builds only)")
endif()
set(VEDIT_SYNTAX_TS_LINK "shared" CACHE STRING "libtree-sitter linkage: shared|static")
set(VEDIT_TREE_SITTER_VERSION "0.25.9" CACHE STRING "exact tree-sitter version (see scripts/versions-qt.env)")
option(VEDIT_SYNTAX_BUILD_TESTS "Build the unit tests (standalone evidence mode)" ON)

if(NOT VEDIT_SYNTAX_TS_LINK MATCHES "^(shared|static)$")
  message(FATAL_ERROR "VEDIT_SYNTAX_TS_LINK must be 'shared' or 'static', got '${VEDIT_SYNTAX_TS_LINK}'")
endif()

# ---- compile-time configuration header (build artifact, never a source edit) ----
# Written by a function so it can run at the END of the top-level configure in a
# product build (the grammar lane is added after this directory). The header, the
# provider test's expectation and the cache value therefore cannot disagree.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")

# The generated directory and the lane directory are handed over through CACHE
# variables on purpose: `cmake_language(DEFER ... CALL ...)` arguments are
# expanded when the deferred call RUNS (in the top-level directory scope), not
# when it is scheduled, so a local variable would arrive empty.
set(VEDIT_SYNTAX_GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated"
    CACHE INTERNAL "syn-provider generated-config directory (build artifact)")
set(VEDIT_SYNTAX_PROVIDER_LANE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
    CACHE INTERNAL "syn-provider lane source directory")

function(vedit_syntax_write_provider_config)
  set(generated_dir "${VEDIT_SYNTAX_GEN_DIR}")
  set(lane_dir "${VEDIT_SYNTAX_PROVIDER_LANE_DIR}")
  set(_grammar "${VEDIT_SYNTAX_GRAMMAR}")
  if("${_grammar}" STREQUAL "")
    message(FATAL_ERROR
      "VEDIT_SYNTAX_GRAMMAR is empty: a product build must name the compiled grammar\n"
      "       (e.g. -DVEDIT_SYNTAX_GRAMMAR=verilog) or include the lane that owns it\n"
      "       (src/syntax/grammars). A build with an unnamed grammar is not the product.")
  endif()
  if(NOT _grammar MATCHES "^(none|external|verilog|systemverilog)$")
    message(FATAL_ERROR
      "VEDIT_SYNTAX_GRAMMAR must be none|external|verilog|systemverilog, got '${_grammar}'")
  endif()
  set(_ts_compiled "false")
  if(VEDIT_SYNTAX_TREESITTER)
    set(_ts_compiled "true")
  endif()
  set(_ts_static "false")
  if(VEDIT_SYNTAX_TS_LINK STREQUAL "static")
    set(_ts_static "true")
  endif()
  file(WRITE "${generated_dir}/grammar_config_generated.h"
"// Generated by CMake from VEDIT_SYNTAX_GRAMMAR / VEDIT_SYNTAX_TREESITTER / VEDIT_SYNTAX_TS_LINK.
// This file is a BUILD ARTIFACT: switching grammar rewrites it, no tracked source changes.
#pragma once
namespace vedit::syntax::generated {
inline constexpr const char *kCompiledGrammar = \"${_grammar}\";
inline constexpr bool kBackendCompiled = ${_ts_compiled};
inline constexpr bool kStaticBackend = ${_ts_static};
}  // namespace vedit::syntax::generated
")
  if(TARGET vedit_syntax_provider_tests)
    target_compile_definitions(vedit_syntax_provider_tests PRIVATE
      "VEDIT_EXPECT_COMPILED_GRAMMAR=\"${_grammar}\"")
  endif()
  message(STATUS
    "syn-provider: baked grammar id='${_grammar}' for ${lane_dir} (owner: ${VEDIT_SYNTAX_GRAMMAR_OWNER})")
endfunction()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  # No grammar lane in a standalone provider build: "none" is the honest default.
  set(VEDIT_SYNTAX_GRAMMAR "none" CACHE STRING
      "Grammar binding id baked into this build (none|external|verilog|systemverilog)")
endif()
# Both cases defer to the end of the directory being processed: in a product
# build that is after the grammar lane set the cache; standalone it is after this
# file created its test target.
cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL vedit_syntax_write_provider_config)

# ---- tree-sitter discovery (reuse the root build's target when present) ----
if(VEDIT_SYNTAX_TREESITTER)
  if(TARGET PkgConfig::TREE_SITTER)
    set(VEDIT_TS_TARGET PkgConfig::TREE_SITTER)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(VEDIT_TREE_SITTER REQUIRED IMPORTED_TARGET "tree-sitter=${VEDIT_TREE_SITTER_VERSION}")
    set(VEDIT_TS_TARGET PkgConfig::VEDIT_TREE_SITTER)
  endif()
endif()

# ---- library ----
set(VEDIT_SYNTAX_PROVIDER_SOURCES
  disabled_provider.cpp
  grammar_binding.cpp
  owner_arbiter.cpp
  syntax_service.cpp
)
if(VEDIT_SYNTAX_TREESITTER)
  list(APPEND VEDIT_SYNTAX_PROVIDER_SOURCES tree_sitter_provider.cpp)
else()
  list(APPEND VEDIT_SYNTAX_PROVIDER_SOURCES tree_sitter_unavailable.cpp)
endif()

add_library(vedit_syntax_provider STATIC ${VEDIT_SYNTAX_PROVIDER_SOURCES})
target_include_directories(vedit_syntax_provider PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}"
  "${VEDIT_SYNTAX_GEN_DIR}"
)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(vedit_syntax_provider PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(VEDIT_SYNTAX_TREESITTER)
  if(VEDIT_SYNTAX_TS_LINK STREQUAL "static")
    # C18: the delivery bundle may link libtree-sitter statically. The library is
    # then an implementation detail of this target; grammar replaceability stays
    # at the compile-time configuration layer (registry + CMake option).
    find_library(VEDIT_TREE_SITTER_STATIC NAMES libtree-sitter.a REQUIRED)
    target_include_directories(vedit_syntax_provider PUBLIC ${VEDIT_TREE_SITTER_INCLUDE_DIRS})
    target_link_libraries(vedit_syntax_provider PUBLIC "${VEDIT_TREE_SITTER_STATIC}")
    message(STATUS "syn-provider: linking ${VEDIT_TREE_SITTER_STATIC} statically (C18)")
  else()
    target_link_libraries(vedit_syntax_provider PUBLIC ${VEDIT_TS_TARGET})
  endif()
endif()

# ---- unit tests (need the tree-sitter backend + the test fixture grammar) ----
if(VEDIT_SYNTAX_BUILD_TESTS AND VEDIT_SYNTAX_TREESITTER)
  # Test-only fixture grammar: tree-sitter-json (MIT, pinned commit). It is NOT a
  # product grammar (src/syn-grammars owns those); it exists so the generic
  # backend, queries, breadcrumbs and incremental sync are exercised for real.
  add_library(vedit_syntax_test_fixture_grammar STATIC tests/fixture/parser.c)
  target_include_directories(vedit_syntax_test_fixture_grammar PRIVATE tests/fixture)

  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture/queries/highlights.scm" _q_highlights)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture/queries/folds.scm" _q_folds)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture/queries/structure.scm" _q_structure)
  file(WRITE "${VEDIT_SYNTAX_GEN_DIR}/fixture_queries_generated.h"
"// Generated from tests/fixture/queries/*.scm (test fixture only).
#pragma once
namespace vedit::syntax::tests::generated {
inline constexpr const char *kHighlights = R\"VEDITQ(${_q_highlights})VEDITQ\";
inline constexpr const char *kFolds = R\"VEDITQ(${_q_folds})VEDITQ\";
inline constexpr const char *kStructure = R\"VEDITQ(${_q_structure})VEDITQ\";
}  // namespace vedit::syntax::tests::generated
")

  add_executable(vedit_syntax_provider_tests
    tests/test_syntax_provider.cpp
    tests/fixture/fixture_binding.cpp
  )
  target_include_directories(vedit_syntax_provider_tests PRIVATE
    "${VEDIT_SYNTAX_GEN_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture"
  )
  target_link_libraries(vedit_syntax_provider_tests PRIVATE
    vedit_syntax_provider
    vedit_syntax_test_fixture_grammar
  )
  if(VEDIT_SYNTAX_TS_LINK STREQUAL "static")
    target_link_libraries(vedit_syntax_provider_tests PRIVATE "${VEDIT_TREE_SITTER_STATIC}")
  else()
    target_link_libraries(vedit_syntax_provider_tests PRIVATE ${VEDIT_TS_TARGET})
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(vedit_syntax_provider_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  target_compile_definitions(vedit_syntax_provider_tests PRIVATE
    VEDIT_EXPECT_TREE_SITTER="${VEDIT_SYNTAX_TREESITTER}"
  )
  # VEDIT_EXPECT_COMPILED_GRAMMAR is added by vedit_syntax_write_provider_config()
  # after the grammar lane has set the cache value.
  add_test(NAME vedit-syntax-provider-tests COMMAND vedit_syntax_provider_tests)
  set_tests_properties(vedit-syntax-provider-tests PROPERTIES TIMEOUT 120)
elseif(VEDIT_SYNTAX_BUILD_TESTS)
  message(STATUS "syn-provider: tests skipped (tree-sitter backend not compiled)")
endif()
