cmake_minimum_required(VERSION 3.20)
project(c15_seam LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# ---------------------------------------------------------------------------
# The grammar registry. This block is the ONLY place in the probe that names a
# concrete grammar: the interface (src/syntax_provider.hpp), the implementation
# (src/tree_sitter_provider.cpp) and every caller (src/main.cpp) are identical
# for every value of SEAM_GRAMMAR. Switching grammars is therefore a
# configuration change, never a source change.
# ---------------------------------------------------------------------------
option(SEAM_TREESITTER "Compile the tree-sitter backed SyntaxProvider" ON)
set(SEAM_GRAMMAR "json" CACHE STRING "Grammar compiled into the seam: json | verilog_placeholder")
set(SEAM_TS_LINK "shared" CACHE STRING "libtree-sitter linkage: shared | static")

if(SEAM_GRAMMAR STREQUAL "json")
  set(GRAMMAR_DIR     "${CMAKE_CURRENT_SOURCE_DIR}/third_party/tree-sitter-json")
  set(GRAMMAR_ID      "json")
  set(GRAMMAR_DISPLAY "tree-sitter-json")
  set(GRAMMAR_SYMBOL  tree_sitter_json)
  set(GRAMMAR_VERSION "254c42a6476413b776221e03982ac8ae159eeb72")
  set(GRAMMAR_LICENSE "MIT")
elseif(SEAM_GRAMMAR STREQUAL "verilog_placeholder")
  set(GRAMMAR_DIR     "${CMAKE_CURRENT_SOURCE_DIR}/grammars/verilog_placeholder")
  set(GRAMMAR_ID      "verilog_placeholder")
  set(GRAMMAR_DISPLAY "verilog_placeholder (minimal Verilog-family grammar)")
  set(GRAMMAR_SYMBOL  tree_sitter_verilog_placeholder)
  set(GRAMMAR_VERSION "generated by tree-sitter-cli 0.27.1")
  set(GRAMMAR_LICENSE "MIT (authored for this probe)")
else()
  message(FATAL_ERROR "SEAM_GRAMMAR must be 'json' or 'verilog_placeholder', got '${SEAM_GRAMMAR}'")
endif()

if(NOT SEAM_TS_LINK MATCHES "^(shared|static)$")
  message(FATAL_ERROR "SEAM_TS_LINK must be 'shared' or 'static', got '${SEAM_TS_LINK}'")
endif()

# ---------------------------------------------------------------------------
# Generate the build-local grammar configuration: identity + logical queries.
# The generated header is a BUILD ARTIFACT; no tracked source file changes when
# SEAM_GRAMMAR changes.
# ---------------------------------------------------------------------------
set(GEN_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN_DIR}")
file(READ "${GRAMMAR_DIR}/queries/highlights.scm" Q_HIGHLIGHTS)
file(READ "${GRAMMAR_DIR}/queries/folds.scm"      Q_FOLDS)
file(READ "${GRAMMAR_DIR}/queries/structure.scm"  Q_STRUCTURE)

file(WRITE "${GEN_DIR}/grammar_config_generated.hpp"
"// Generated by CMake from SEAM_GRAMMAR=${SEAM_GRAMMAR}. Do not edit.
#pragma once
#define SEAM_GRAMMAR_SYMBOL ${GRAMMAR_SYMBOL}
namespace seam::generated {
inline constexpr const char *kGrammarId      = \"${GRAMMAR_ID}\";
inline constexpr const char *kGrammarDisplay = \"${GRAMMAR_DISPLAY}\";
inline constexpr const char *kGrammarVersion = \"${GRAMMAR_VERSION}\";
inline constexpr const char *kGrammarLicense = \"${GRAMMAR_LICENSE}\";
inline constexpr const char *kHighlightQuery = R\"SEAMQ(${Q_HIGHLIGHTS})SEAMQ\";
inline constexpr const char *kFoldQuery      = R\"SEAMQ(${Q_FOLDS})SEAMQ\";
inline constexpr const char *kStructureQuery = R\"SEAMQ(${Q_STRUCTURE})SEAMQ\";
}  // namespace seam::generated
")

# ---------------------------------------------------------------------------
# Sources. The caller and the interface never vary with SEAM_GRAMMAR; only the
# presence of the tree-sitter implementation depends on SEAM_TREESITTER.
# ---------------------------------------------------------------------------
if(SEAM_TREESITTER)
  add_library(grammar_${SEAM_GRAMMAR} STATIC "${GRAMMAR_DIR}/src/parser.c")
  target_include_directories(grammar_${SEAM_GRAMMAR} PUBLIC "${GRAMMAR_DIR}/src")
  set(SEAM_CORE_SOURCES src/main.cpp src/disabled_provider.cpp src/tree_sitter_provider.cpp
      src/grammar_registry.cpp)
else()
  set(SEAM_CORE_SOURCES src/main.cpp src/disabled_provider.cpp src/builtin_fallback.cpp)
endif()

add_executable(c15-seam ${SEAM_CORE_SOURCES})
target_include_directories(c15-seam PRIVATE src "${GEN_DIR}")

if(SEAM_TREESITTER)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(TREESITTER REQUIRED tree-sitter)
  target_include_directories(c15-seam PRIVATE ${TREESITTER_INCLUDE_DIRS})
  target_link_libraries(c15-seam PRIVATE grammar_${SEAM_GRAMMAR})

  if(SEAM_TS_LINK STREQUAL "shared")
    target_link_directories(c15-seam PRIVATE ${TREESITTER_LIBRARY_DIRS})
    target_link_libraries(c15-seam PRIVATE ${TREESITTER_LIBRARIES})
  else()
    find_library(TREESITTER_STATIC_PATH NAMES libtree-sitter.a REQUIRED)
    message(STATUS "linking ${TREESITTER_STATIC_PATH} statically")
    target_link_libraries(c15-seam PRIVATE "${TREESITTER_STATIC_PATH}")
  endif()
  target_compile_definitions(c15-seam PRIVATE SEAM_TREESITTER=1)
else()
  target_compile_definitions(c15-seam PRIVATE SEAM_TREESITTER=0)
endif()

install(TARGETS c15-seam RUNTIME DESTINATION bin)
